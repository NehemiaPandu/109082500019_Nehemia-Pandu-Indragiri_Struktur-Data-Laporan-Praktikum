## Guided 

### 1. Mahasiswa

```C++
#ifndef Mahasiswa_H_Included
#define Mahasiswa_H_Included

struct mahasiswa {
    char nim[10];
    int nilai1, nilai2;
};

void inputMhs (mahasiswa &m);
float rata2 (mahasiswa m);
#endif
```
```C++
#include <iostream>
#include "mahasiswa.h"

using namespace std;

void inputMhs(mahasiswa &m) {
    cout << "Input NIM = ";
    cin >> (m).nim;
    cout << "Input Nilai 1 = ";
    cin >> (m).nilai1;
    cout << "Input Nilai 2 = ";
    cin >> (m).nilai2;
}

float rata2(mahasiswa m) {
    return float(m.nilai1 + m.nilai2) / 2;
    
}
```
```C++
#include <iostream>
#include "mahasiswa.h"

using namespace std;

int main(){
    mahasiswa mhs;
    inputMhs (mhs);
    cout << "Rata-rata: " << rata2 (mhs);
    
    return 0;
}
```
